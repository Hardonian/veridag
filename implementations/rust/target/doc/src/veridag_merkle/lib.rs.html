1//! BMH-1: Binary Merkle Hash-map commitment (spec 12-state).
2//!
3//! Deterministic authenticated commitment to a set of `(ObjectId -> Object)`
4//! pairs. Leaves sorted by ObjectId bytewise; canonical encoding throughout.
5

9use thiserror::Error;
10use veridag_crypto::hash;
11use veridag_protocol_types::{Hash, ObjectId};

13/// Errors from proof verification.
14#[derive(Debug, Error, Clone, PartialEq, Eq)]
15pub enum MerkleError {
16    /// Proof did not recompute to the expected root.
17    #[error("invalid proof")]
18    InvalidProof,
19    /// Proof encoding was malformed.
20    #[error("malformed proof")]
21    MalformedProof,
22}
23
24/// A BMH-1 inclusion proof: sibling hashes from leaf to root, with a bit
25/// indicating whether the proven node is the left (false) or right (true) child.
26#[derive(Clone, Debug, PartialEq, Eq)]
27pub struct InclusionProof {
28    /// Sibling hashes, leaf-first.
29    pub siblings: Vec<Hash>,
30    /// For each level, true if the proven node is the right child.
31    pub right: Vec<bool>,
32}
33
34/// Compute the leaf hash of an object given its canonical encoding.
35pub fn leaf_hash(object_bytes: &[u8]) -> Hash {
36    hash("VERIDAG_BMH_LEAF_V1", object_bytes)
37}
38
39fn node_hash(l: &Hash, r: &Hash) -> Hash {
40    let mut buf = [0u8; 64];
41    buf[..32].copy_from_slice(l);
42    buf[32..].copy_from_slice(r);
43    hash("VERIDAG_BMH_NODE_V1", &buf)
44}
45
46/// The canonical empty-tree root.
47pub fn empty_root() -> Hash {
48    let mut buf = [0u8; 2];
49    hash("VERIDAG_BMH_NODE_V1", &{
50        buf[0] = 0;
51        buf[1] = 0;
52        buf
53    })
54}

56/// Compute the BMH-1 root of a set of `(ObjectId, object_bytes)` leaves.
57///
58/// Leaves are sorted by ObjectId bytewise. Duplicate ids are rejected by the
59/// caller before calling this function.
60pub fn root(leaves: &[(ObjectId, Hash)]) -> Hash {
61    if leaves.is_empty() {
62        return empty_root();
63    }
64    let mut sorted: Vec<(ObjectId, Hash)> = leaves.to_vec();
65    sorted.sort_by_key(|(id, _)| *id);
66    let mut level: Vec<Hash> = sorted.into_iter().map(|(_, h)| h).collect();
67    while level.len() > 1 {
68        let mut next = Vec::with_capacity(level.len().div_ceil(2));
69        let mut i = 0;
70        while i < level.len() {
71            if i + 1 < level.len() {
72                next.push(node_hash(&level[i], &level[i + 1]));
73            } else {
74                // odd leaf promoted unchanged
75                next.push(level[i]);
76            }
77            i += 2;
78        }
79        level = next;
80    }
81    level[0]
82}

84/// Generate an inclusion proof for the object at `index` in the sorted leaf list.
85pub fn prove(leaves: &[(ObjectId, Hash)], index: usize) -> Option<InclusionProof> {
86    if leaves.is_empty() || index >= leaves.len() {
87        return None;
88    }
89    let mut sorted: Vec<(ObjectId, Hash)> = leaves.to_vec();
90    sorted.sort_by_key(|(id, _)| *id);
91    let mut level: Vec<Hash> = sorted.iter().map(|(_, h)| *h).collect();
92    let mut idx = index;
93    let mut siblings = Vec::new();
94    let mut right = Vec::new();
95    while level.len() > 1 {
96        let sibling_idx = if idx.is_multiple_of(2) {
97            idx + 1
98        } else {
99            idx - 1
100        };
101        // If there is no sibling (odd node promoted), record the node's own hash
102        // as sibling with a flag; verification handles promotion.
103        if sibling_idx < level.len() {
104            siblings.push(level[sibling_idx]);
105            right.push(!idx.is_multiple_of(2));
106        } else {
107            siblings.push(level[idx]);
108            right.push(false); // promoted: treat as left, sibling is self
109        }
110        idx /= 2;
111        let mut next = Vec::with_capacity(level.len().div_ceil(2));
112        let mut i = 0;
113        while i < level.len() {
114            if i + 1 < level.len() {
115                next.push(node_hash(&level[i], &level[i + 1]));
116            } else {
117                next.push(level[i]);
118            }
119            i += 2;
120        }
121        level = next;
122    }
123    Some(InclusionProof { siblings, right })
124}
125
126/// Verify an inclusion proof for `leaf` against `expected_root`.
127pub fn verify(
128    leaf: &Hash,
129    proof: &InclusionProof,
130    expected_root: &Hash,
131) -> Result<(), MerkleError> {
132    if proof.siblings.len() != proof.right.len() {
133        return Err(MerkleError::MalformedProof);
134    }
135    let mut acc = *leaf;
136    for (sib, is_right) in proof.siblings.iter().zip(proof.right.iter()) {
137        acc = if *is_right {
138            node_hash(sib, &acc)
139        } else if sib == &acc {
140            // promotion level: node carried up unchanged
141            acc
142        } else {
143            node_hash(&acc, sib)
144        };
145    }
146    if &acc == expected_root {
147        Ok(())
148    } else {
149        Err(MerkleError::InvalidProof)
150    }
151}

153#[cfg(test)]
154mod tests {
155    use super::*;
156
157    fn id(b: u8) -> ObjectId {
158        ObjectId([b; 32])
159    }
160    fn h(b: u8) -> Hash {
161        hash("VERIDAG_BMH_LEAF_V1", &[b])
162    }
163
164    #[test]
165    fn empty_tree_root_is_deterministic() {
166        assert_eq!(root(&[]), empty_root());
167    }
168
169    #[test]
170    fn root_is_order_independent_for_input() {
171        let l1 = vec![(id(1), h(1)), (id(2), h(2))];
172        let l2 = vec![(id(2), h(2)), (id(1), h(1))];
173        assert_eq!(root(&l1), root(&l2), "root sorts leaves canonically");
174    }
175
176    #[test]
177    fn root_changes_with_leaves() {
178        let a = root(&[(id(1), h(1))]);
179        let b = root(&[(id(1), h(9))]);
180        assert_ne!(a, b);
181    }
182
183    #[test]
184    fn inclusion_proof_verifies() {
185        let leaves: Vec<(ObjectId, Hash)> = (0u8..7).map(|b| (id(b), h(b))).collect();
186        let r = root(&leaves);
187        for i in 0..leaves.len() {
188            let proof = prove(&leaves, i).unwrap();
189            let mut sorted = leaves.clone();
190            sorted.sort_by_key(|(id, _)| *id);
191            assert!(
192                verify(&sorted[i].1, &proof, &r).is_ok(),
193                "proof {i} must verify"
194            );
195        }
196    }
197
198    #[test]
199    fn bad_proof_rejected() {
200        let leaves: Vec<(ObjectId, Hash)> = (0u8..4).map(|b| (id(b), h(b))).collect();
201        let r = root(&leaves);
202        let proof = prove(&leaves, 0).unwrap();
203        assert_eq!(verify(&h(99), &proof, &r), Err(MerkleError::InvalidProof));
204    }
205}
